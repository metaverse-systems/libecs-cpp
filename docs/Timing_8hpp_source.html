58 bool ShouldUpdate(std::chrono::microseconds now)
59 {
60 if(this->updateInterval == 0)
61 return true;
62
63 int64_t current_time = now.count();
64 int64_t elapsed = current_time - this->lastUpdateTime;
65 if(elapsed < 0)
66 elapsed = 0;
67
68 if(elapsed >= this->updateInterval)
69 {
70 // Remainder-carry: advance by the largest multiple of the interval that fits, so the
71 // average rate is exact and drift-free under jittered sampling.
72 int64_t intervals = elapsed / this->updateInterval;
73
74 // MAX_CATCHUP clamp: if stalled for too long, snap forward instead of catching up
75 // (prevents a burst after suspend or a long pause).
76 if(intervals > static_cast<int64_t>(MAX_CATCHUP))
77 this->lastUpdateTime = current_time;
