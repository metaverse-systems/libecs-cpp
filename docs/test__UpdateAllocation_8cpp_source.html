162 // Builds a world of `count` systems whose first system removes the last
163 // one when armed, runs a warm-up pass, then returns the allocation count
164 // of one measured pass that performs the removal.
165 std::size_t removalPassAllocations(int count)
166 {
167 ecs::Manager manager;
168 auto container = manager.Container("allocation-removal");
169 auto *first = static_cast<RemovingSystem *>(container->System(
170 std::make_unique<RemovingSystem>(longHandle(0), longHandle(count - 1))));
171 for (int i = 1; i < count; i++)
172 container->System(std::make_unique<CountingSystem>(longHandle(i)));
173
174 container->Update();
175
176 first->armed = true;
177 allocationCount = 0;
178 measuring = true;
179 container->Update();
180 measuring = false;
181 std::size_t result = allocationCount.load();
182
183 REQUIRE_FALSE(container->Systems.contains(longHandle(count - 1)));
184 return result;
185 }
