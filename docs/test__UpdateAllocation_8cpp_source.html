161 // Builds a world of `count` systems whose first system removes the last
162 // one when armed, runs a warm-up pass, then returns the allocation count
163 // of one measured pass that performs the removal.
164 std::size_t removalPassAllocations(int count)
165 {
166 ecs::Manager manager;
167 auto container = manager.Container("allocation-removal");
168 auto *first = static_cast<RemovingSystem *>(container->System(
169 std::make_unique<RemovingSystem>(longHandle(0), longHandle(count - 1))));
170 for (int i = 1; i < count; i++)
171 container->System(std::make_unique<CountingSystem>(longHandle(i)));
172
173 container->Update();
174
175 first->armed = true;
176 allocationCount = 0;
177 measuring = true;
178 container->Update();
179 measuring = false;
180 std::size_t result = allocationCount.load();
181
182 REQUIRE_FALSE(container->Systems.contains(longHandle(count - 1)));
183 return result;
184 }
