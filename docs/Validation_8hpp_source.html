63 // Returns the destination object of a message, or throws.
64 inline const nlohmann::json &messageDestination(const nlohmann::json &message, Caller caller)
65 {
66 if(!message.is_object()) failNotObject(caller, message);
67 auto found = message.find("destination");
68 if(found == message.end()) failDestinationMissing(caller);
69 if(!found->is_object()) failDestinationType(caller, *found);
70 return *found;
71 }

73 inline const std::string &destinationField(const nlohmann::json &destination, const char *field, Caller caller)
74 {
75 auto found = destination.find(field);
76 if(found == destination.end()) failFieldMissing(caller, field);
77 if(!found->is_string()) failFieldType(caller, field, *found);
78 const std::string &name = found->get_ref<const std::string &>();
79 if(name.empty()) failFieldEmpty(caller, field);
80 return name;
81 }
