142TEST_CASE("DeltaTimeGet is stable within an update and carries its remainder", "[Compatibility]")
143{
144 ecs::ManualClock clock(micros(1000000));
145 ecs::Manager manager;
146 auto container = manager.Container("old-delta");
147 auto *system = static_cast<OldStyleSystem *>(container->System(std::make_unique<OldStyleSystem>()));
148 container->ClockSet(&clock);
149 system->Timing.SetInterval(micros(33333));
150
151 constexpr int updates = 301;
152 const micros begin = clock.Now();
153 for (int i = 0; i < updates; ++i)
154 {
155 clock.Advance(micros(33333));
156 container->Update();
157 }
158
159 REQUIRE(system->stable);
160 REQUIRE(system->values.size() == static_cast<size_t>(updates));
161
162 // From the second update on the whole milliseconds add up to the clock's total, within one.
163 int64_t sum = 0;
164 for (size_t i = 1; i < system->values.size(); ++i)
165 sum += system->values[i];
166 const int64_t clockMs = (33333LL * (updates - 1)) / 1000;
167 REQUIRE(sum >= clockMs - 1);
168 REQUIRE(sum <= clockMs + 1);
169 REQUIRE(clock.Now() - begin == micros(33333LL * updates));
170
171 // A step of 33333 us reads mostly 33 ms, with a 34 about every third update.
172 int thirtyFour = 0;
173 for (size_t i = 1; i < system->values.size(); ++i)
174 {
175 REQUIRE((system->values[i] == 33 || system->values[i] == 34));
176 if (system->values[i] == 34)
177 thirtyFour++;
178 }
179 REQUIRE(thirtyFour > 0);
180 REQUIRE(thirtyFour < updates / 2);
181 const std::vector<uint32_t> expected = {33, 33, 33, 34, 33, 33, 34};
182 for (size_t i = 0; i < expected.size(); ++i)
