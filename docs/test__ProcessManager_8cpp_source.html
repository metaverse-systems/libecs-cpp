69 // Runs fn on a worker thread and waits for it. When the time is up the worker is joined (never
70 // detached) and the test fails.
71 template <typename Fn>
72 void withTimeout(Fn fn, std::chrono::seconds limit)
73 {
74 std::packaged_task<void()> task(std::move(fn));
75 std::future<void> future = task.get_future();
76 std::thread worker([&task] { task(); });
77 const bool finished = future.wait_for(limit) == std::future_status::ready;
78 worker.join();
79 if(!finished) FAIL("timed out after " << limit.count() << " s");
80 future.get();
81 }
