40 // A system that records what it is told about time on each update.
41 class Probe : public ecs::System
42 {
43 public:
44 explicit Probe(const std::string &handle = "probe") : System(handle) {}
45
46 nlohmann::json Export() const override
47 {
48 return nlohmann::json::object();
49 }
50
51 void Update() override
52 {
53 this->updates++;
54 const micros first = this->ElapsedGet();
55 const double seconds = this->ElapsedSecondsGet();
56 for (int i = 1; i < this->reads; ++i)
57 {
58 if (this->ElapsedGet() != first)
59 this->agree = false;
60 if (this->ElapsedSecondsGet() != seconds)
61 this->agree = false;
62 }
63 if (this->reads > 0)
64 {
65 this->lastElapsed = first;
66 this->lastSeconds = seconds;
67 this->total += first;
68 if (this->keep)
69 this->values.push_back(first);
70 }
71 if (this->watch != nullptr)
72 this->instants.push_back(this->watch->Now());
73 if (this->onUpdate)
74 this->onUpdate();
75 }
76
77 size_t TimerTotal() const
78 {
79 return this->timers.size();
80 }
81
82 int updates = 0;
83 // How many times Update() reads the elapsed time; zero means it never asks.
84 int reads = 1;
85 bool agree = true;
86 bool keep = false;
87 micros lastElapsed{-1};
88 double lastSeconds = -1;
89 micros total{0};
90 std::vector<micros> values;
91 const ecs::Clock *watch = nullptr;
92 std::vector<micros> instants;
93 std::function<void()> onUpdate;
94 };

115 // Updates a lone system with a step per update and checks the elapsed values add up to the
116 // time the clock moved between the first and the last update.
117 template <typename StepFor>
118 void sumCheck(micros start, int64_t updates, StepFor stepFor)
119 {
120 ecs::ManualClock clock(start);
121 Probe probe;
122 probe.Timing.SetInterval(us(0));
123 probe.ClockSet(&clock);
124
125 probe.UpdateSystem();
126 const micros firstInstant = clock.Now();
127 micros sum{0};
128 for (int64_t i = 0; i < updates; ++i)
129 {
130 clock.Advance(stepFor(i));
131 probe.UpdateSystem();
132 sum += probe.lastElapsed;
133 }
134 REQUIRE(probe.agree);
135 const int64_t difference = (sum - (clock.Now() - firstInstant)).count();
136 REQUIRE(std::abs(difference) <= 1);
137 }
