99 void Container::walkFinish()
100 {
101 this->walkDepth--;
102 if(this->walkDepth == 0 && this->orderHasGaps)
103 {
104 std::erase_if(this->system_order, [](const SystemSlot &slot) { return slot.system == nullptr; });
105 this->orderHasGaps = false;
106 }
107 if(this->walkDepth == 0 && !this->retiredSystems.empty())
108 {
109 // Release outside any walk, so a destructor that changes systems sees a settled container.

115 void Container::systemRetire(std::unique_ptr<ecs::System> system)
116 {
117 ecs::System *ptr = system.get();
118 if(this->walkDepth > 0)
119 {
120 for(auto &slot : this->system_order)
121 {
122 if(slot.system == ptr)
123 {
124 slot.system = nullptr;
125 this->orderHasGaps = true;
126 }
127 }
128 }
129 else

131 std::erase_if(this->system_order, [ptr](const SystemSlot &slot) { return slot.system == ptr; });
132 }
133

135 {

137 system->removed = true;
138 this->retiredSystems.push_back(std::move(system));

142 ecs::System *Container::System(std::unique_ptr<ecs::System> system)
143 {
144 system->Container = this;
145 system->Components = &(this->Components);
146 const std::string handle = system->Handle;
147 ecs::System *ptr = system.get();
148
149 auto existing = this->Systems.find(handle);
150 if(existing != this->Systems.end())
151 {
152 std::unique_ptr<ecs::System> old = std::move(existing->second);
153 for(auto &slot : this->system_order)
154 {
155 if(slot.system == old.get()) slot.system = ptr;
156 }
157 existing->second = std::move(system);
158 this->system_order.push_back(SystemSlot{handle, ptr});
159 this->systemRetire(std::move(old));
160 return ptr;
161 }
162
163 this->Systems.emplace(handle, std::move(system));
164 this->system_order.push_back(SystemSlot{handle, ptr});
165 return ptr;
166 }
