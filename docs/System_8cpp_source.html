31 void System::updateSystem(std::chrono::microseconds now)
32 {
33 this->elapsedMeasure(now);
34 this->mailboxDrain();
35 this->timerWalkDepth++;
36 try
37 {
38 for(size_t i = 0, count = this->timers.size(); i < count && !this->removed; i++)
39 {
40 // Callbacks may add or cancel timers, so never hold a reference across fire().
41 if(this->timers[i].discarded || !this->timers[i].due(now)) continue;
42 if(!this->timers[i].Repeat)
43 {
44 this->timers[i].discarded = true;
45 this->timersDiscarded = true;
46 }
47 this->timers[i].fire();
48 }
49 }
50 catch(...)

52 this->timerWalkFinish();
53 throw;
54 }
55 this->timerWalkFinish();
56
57 if(!this->removed) this->Update();
58 }

107 void System::elapsedMeasure(std::chrono::microseconds now)
108 {
109 if(!this->updated)
110 {
111 // Nothing earlier to measure from: report the configured interval.
112 this->elapsed = this->Timing.GetInterval();
113 this->previousUpdate = now;
114 this->updated = true;
115 }
116 else
117 {
118 // The clock never goes backwards; an instant that did would count as no time passed.
119 this->elapsed = std::max(std::chrono::microseconds(0), now - this->previousUpdate);
120 this->previousUpdate = std::max(this->previousUpdate, now);
121 }
122
123 const std::chrono::microseconds total = this->millisecondCarry + this->elapsed;
124 const int64_t whole = total / std::chrono::milliseconds(1);
125 this->millisecondCarry = total % std::chrono::milliseconds(1);
126 this->elapsedMilliseconds = static_cast<uint32_t>(std::min<int64_t>(whole, UINT32_MAX));
127 }
