31 // Collects every (message, level) pair a world delivers to its destination.
32 struct Recorder
33 {
34 using Line = std::pair<std::string, std::string>;
35
36 void record(const std::string &message, const std::string &level)
37 {
38 std::lock_guard<std::mutex> lock(this->lock);
39 this->lines.emplace_back(message, level);
40 }
41
42 std::vector<Line> snapshot() const
43 {
44 std::lock_guard<std::mutex> lock(this->lock);
45 return this->lines;
46 }
47
48 size_t count() const { return this->snapshot().size(); }
49
50 bool contains(const std::string &message, const std::string &level) const
51 {
52 const auto all = this->snapshot();
53 return std::find(all.begin(), all.end(), Line(message, level)) != all.end();
54 }
55
56 // Installs the recorder as the destination of the world. The recorder must outlive the world's use of it.
57 void install(ecs::Container *world)
58 {
59 world->LoggerSet([this](const std::string &message, const std::string &level) { this->record(message, level); });
60 }
61
62 mutable std::mutex lock;
63 std::vector<Line> lines;
64 };
